331proof fn chain_of_optional_defaulted() {
332    use crate::combinators::disjoint::disjointness_lemmas;
333    use super::disjoint::asn1_disjointness_lemmas;
334
335    broadcast use disjointness_lemmas;
336    broadcast use asn1_disjointness_lemmas;
337
338    assert(MY_FMT().safe_inv());
339    assert(MY_FMT().sound_inv());
340    assert(MY_FMT().unambiguous());
341    assert(MY_FMT().nonmal_inv());
342
343    assert(MY_FMT2().safe_inv());
344    assert(MY_FMT2().sound_inv());
345    assert(MY_FMT2().unambiguous());
346    assert(MY_FMT2().nonmal_inv());
347
348    #[verusfmt::skip]
349    let fmt =
350        REQUIRED (INTEGER,
351        DEFAULT  (BOOLEAN, false,
352        REQUIRED (BIT_STRING,
353        OPTIONAL (OCTET_STRING,
354        DEFAULT  (INTEGER8, 0,
355        OPTIONAL (UTC_TIME, UTF8_STRING))))));
356
357    assert(fmt.safe_inv());
358    assert(fmt.sound_inv());
359    assert(fmt.unambiguous());
360
361    #[verusfmt::skip]
362    let fmt1 =
363        REQUIRED  (INTEGER,
364        DEFAULT   (EXPLICIT (0, INTEGER16), 10,
365        OPTIONAL  (IMPLICIT (1, INTEGER16),
366        OPTIONAL  (IMPLICIT (2, INTEGER16),
367        DEFAULT   (EXPLICIT (3, INTEGER16), 0,
368        OPTIONAL  (UTC_TIME, UTF8_STRING))))));
369
370    assert(fmt1.safe_inv());
371    assert(fmt1.sound_inv());
372    assert(fmt1.unambiguous());
373
374    #[verusfmt::skip]
375    let fmt2 = CHOICE(
376        IMPLICIT (0, NULL),     CHOICE(
377        IMPLICIT (1, NULL),     CHOICE(
378        IMPLICIT (2, INTEGER8), CHOICE(
379        EXPLICIT (3, INTEGER8), CHOICE(
380        OCTET_STRING, UTF8_STRING)))));
381
382    assert(fmt2.safe_inv());
383    assert(fmt2.sound_inv());
384    assert(fmt2.unambiguous());
385}
