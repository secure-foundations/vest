125#[derive(Copy)]
126pub struct ASN1Fmt<Content, const DER: bool = true>(pub Tag, pub Content);
127
128impl<Content: Clone, const DER: bool> Clone for ASN1Fmt<Content, DER> {
129    fn clone(&self) -> (cloned: Self)
130        ensures
131            call_ensures(Content::clone, (&self.1,), cloned.1),
132    {
133        ASN1Fmt(self.0, self.1.clone())
134    }
135}

137/// ASN.1 BOOLEAN format.
138///
139/// When `DER = true` (the default), this is the canonical DER form:
140/// FALSE = `0x00`, TRUE = `0xFF`.
141///
142/// When `DER = false`, this is the more permissive BER form:
143/// FALSE = `0x00`, TRUE = any non-zero byte.
144#[derive(Clone, Copy)]
145pub struct BoolFmt<const DER: bool = true>;
146
147/// Convenience type alias for the BER variant of ASN.1 BOOLEAN.
148pub type BerBoolFmt = BoolFmt<false>;

157pub const DerBoolFmt: BoolFmt<true> = BoolFmt;

181/// ASN.1 definite length format.
182///
183/// When `DER = true` (the default), only the canonical DER definite form is
184/// accepted/produced.
185///
186/// When `DER = false`, the parser/serializer is BER-permissive over short and long
187/// definite forms, without minimality constraints.
188#[derive(Clone, Copy)]
189pub struct LengthFmt<const DER: bool = true>;
190
191/// BER length, including the indefinite form (`0x80`).
192#[derive(StructuralEq, Copy, Clone, PartialEq, Eq, Debug)]
193pub enum BerLength {

196}

241/// ASN.1 tag format combinator.

252/// ASN.1 OCTET STRING contents format (primitive).
253///
254/// TODO: support indefinite (constructed) forms:
255///
256/// ### Example
257///
258/// 24 80 04 03 42 45 52 24 80 04 01 2D 04 05 52 55 4C 45 53 00 00 00 00
259/// │  │  │  │  └───────┘ │  │  │  │  │  │  │  └──────────────┘ │     │
260/// │  │  │  │  "BER"     │  │  │  │  │  │  └─ Len: 5 "RULES"   │     └─ Outer EOC
261/// │  │  │  └─ Len: 3    │  │  │  │  │  └─ Primitive           └─ Inner EOC
262/// │  │  └─ Primitive    │  │  │  │  └─ "-"
263/// │  └─ Indefinite      │  │  │  └─ Len: 1
264/// └─ Outer Outer        │  │  └─ Primitive
265///                       │  └─ Indefinite
266///                       └─ Inner Constructed
267///
268/// [TAG: 24] Constructed OCTET STRING (Indefinite Length)
269///  │
270///  ├── [TAG: 04] Primitive OCTET STRING (Definite Length: 3)
271///  │      └── Value: "BER" (Hex: 42 45 52)
272///  │
273///  ├── [TAG: 24] Constructed OCTET STRING (Indefinite Length)
274///  │      │
275///  │      ├── [TAG: 04] Primitive OCTET STRING (Definite Length: 1)
276///  │      │      └── Value: "-" (Hex: 2D)
277///  │      │
278///  │      ├── [TAG: 04] Primitive OCTET STRING (Definite Length: 5)
279///  │      │      └── Value: "RULES" (Hex: 52 55 4C 45 53)
280///  │      │
281///  │      └── [TAG: 00] End-of-Contents (EOC) Marker (Hex: 00 00)
282///  │             └── Meaning: Closes the Inner Constructed String
283///  │
284///  └── [TAG: 00] End-of-Contents (EOC) Marker (Hex: 00 00)
285///         └── Meaning: Closes the Outer Constructed String
286pub type OctetStringFmt = Tail;
287

289pub const OctetStringFmt: Tail = Tail;
