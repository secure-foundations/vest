54impl<'a> TeletexString<'a> {
55    #[verifier::type_invariant]
56    spec fn wf(&self) -> bool {
57        self.deep_view().wf()
58    }
59
60    pub fn new(inner: &'a str) -> (res: Self)
61        requires
62            is_valid_teletex_string_spec(vstd::utf8::encode_utf8(inner.deep_view())),
63        ensures
64            res.deep_view() == (TeletexStringSpec { inner: inner.deep_view() }),
65    {
66        TeletexString { inner }
67    }
68
69    pub fn inner(&self) -> (res: &'a str)
70        ensures
71            res.deep_view() == self.deep_view().inner,
72    {
73        self.inner
74    }

107/// Current TeletexString validity predicate.
108///
109/// This currently accepts every byte sequence: Vest preserves TeletexString
110/// payloads but does not yet validate the legacy character repertoire.
111pub open spec fn is_valid_teletex_string_spec(bytes: Seq<u8>) -> bool {
112    true
113}
114
115/// Checks the current TeletexString validity predicate.
116///
117/// Because repertoire validation is not implemented, this currently returns
118/// `true` for every input.
119pub fn is_valid_teletex_string(_bytes: &[u8]) -> (res: bool)
120    ensures
121        res == is_valid_teletex_string_spec(_bytes.deep_view()),
122{
123    true
124}

126type TeletexStringFmt = Mapped<
127    Refined<Tail, PredFnSpec<Seq<u8>>>,
128    FnSpecMapper<Seq<u8>, TeletexStringSpec>,
129>;
130
131pub open spec fn teletexstring_fmt() -> TeletexStringFmt {
132    Mapped {
133        inner: Refined(
134            Tail,
135            |bytes: Seq<u8>| is_valid_teletex_string_spec(bytes) && vstd::utf8::valid_utf8(bytes),
136        ),
137        mapper: (
138            |bytes: Seq<u8>| TeletexStringSpec { inner: vstd::utf8::decode_utf8(bytes) },
139            |s: TeletexStringSpec| vstd::utf8::encode_utf8(s.inner),
140        ),
141    }
142}

251impl<'i> Parser<&'i [u8]> for super::TeletexStringFmt {
252    type PT = TeletexString<'i>;
253
254    fn parse(&self, ibuf: &&'i [u8]) -> PResult<Self::PT> {
255        let (n, bytes) = Tail.parse(ibuf)?;
256        if !is_valid_teletex_string(bytes) {
257            Err(ParseError::custom("Invalid TeletexString"))
258        } else if !is_valid_utf8(bytes) {
259            Err(ParseError::custom("Invalid UTF-8"))
260        } else {
261            let inner = utf8_from_bytes_unchecked(bytes);
262            Ok((n, TeletexString::new(inner)))
263        }
264    }
265}
