1//! Verified nominal wrappers for ASN.1 schemas.
2//!
3//! [`impl_der!`](crate::impl_der) and [`impl_ber!`](crate::impl_ber) turn a schema
4//! built from combinators into a named format with its own specifications,
5//! proofs, and executable APIs. `vest_asn1` emits these macro invocations.
6//!
7//! # What the caller supplies
8//!
9//! The invocation names the format kind, value ownership, nominal format type,
10//! inner combinator type, specification value type, and executable value type:
11//!
12//! ```rust,ignore
13//! vest_lib::impl_der!(
14//!     tag(constructed), borrowed,
15//!     HEADER, HEADER__, HeaderSpec, Header,
16//!     HeaderForward, HeaderReverse
17//! );
18//! ```
19//!
20//! `borrowed` means the executable type is `Header<'i>`; pass just `Header`.
21//! `owned` takes a complete type with no input lifetime, such as `bool` or `Vec<u8>`.
22//! The optional final arguments name the forward and reverse mappers of a
23//! structural `Mapped<_, BiMap<_, _>>`. Its specification value type supplies
24//! `lemma_into_from` for the mapper's losslessness proof. Omit these arguments
25//! for primitive formats and named aliases.
26//!
27//! The format must provide a `schema()` constructor usable in both specification
28//! and executable code, and `lemma_schema_unambiguous` proving the inner format's
29//! `unambiguous()` condition. Certified start domains also require that lemma to
30//! equate the inner and nominal `asn1_start()` domains, as in the example below.
31//!
32//! # Example: a retaggable DER BOOLEAN
33//!
34//! This is a small version of the code emitted for `Flag ::= BOOLEAN`:
35//!
36//! ```rust,ignore
37//! use vest_lib::asn1::*;
38//! use vest_lib::asn1::der_ord::DerState;
39//! #[cfg(verus_keep_ghost)]
40//! use vest_lib::asn1::disjoint::{
41//!     asn1_start_exact_uint, lemma_asn1_start_exact_uint, HasAsn1Start,
42//! };
43//! use vest_lib::prelude::*;
44//!
45//! verus! {
46//! #[derive(Clone, Copy)]
47//! pub struct FlagFmt(pub Class, pub u64);
48//!
49//! impl FlagFmt {
50//!     pub const Fmt: Self = Self(Class::Universal, 1);
51//!
52//!     #[verifier::allow_in_spec]
53//!     pub const fn schema() -> BoolTlvFmt
54//!         returns BOOLEAN,
55//!     {
56//!         BOOLEAN
57//!     }
58//!
59//!     proof fn lemma_schema_unambiguous(&self)
60//!         ensures
61//!             self.spec_inner().unambiguous(),
62//!             self.spec_inner().asn1_start() == self.asn1_start(),
63//!     {
64//!         reveal(asn1_start_exact_uint);
65//!         lemma_asn1_start_exact_uint(self.0, false, self.1);
66//!     }
67//! }
68//! } // verus!
69//!
70//! vest_lib::impl_der!(tag(primitive), owned, FlagFmt, BoolTlvFmt, bool, bool);
71//! // FlagFmt::Fmt.parse(...), .prepare(...), .serialize(...), and .length(...)
72//! // are now available, together with the proof traits.
73//! ```
74//!
75//! # Format kinds
76//!
77//! A named ASN.1 format needs two pieces of tagging information:
78//! (1). whether its outer tag can be replaced (for IMPLICIT retagging);
79//! and (2). which tags (start_domain) are allowed in a successful parse.
80//! The latter lets Vest prove that CHOICE branches, or an OPTIONAL/DEFAULT field and its
81//! continuation, are disjoint without unfolding their schemas. The format kind
82//! selects the `Retaggable` implementation and the `HasAsn1Start` certificate.
83//!
84//! An ASN.1 identifier carries a class, a primitive/constructed bit, and a tag
85//! number. For example, `BOOLEAN` has universal primitive tag 1 (`01`), while
86//! `SEQUENCE(...)` has universal constructed tag 16 (`30`). Identifier octets
87//! below are hexadecimal. Vest supports:
88//!
89//! - **IMPLICIT tagging:** replace the outer class and number, preserving the
90//!   content's primitive/constructed form. `IMPLICIT(0, BOOLEAN)` starts with
91//!   context-specific primitive tag 0 (`80`), not the BOOLEAN identifier `01`.
92//! - **EXPLICIT tagging:** add a constructed TLV around the complete inner TLV.
93//!   `EXPLICIT(0, BOOLEAN)` starts with `a0` and retains the inner `01` identifier.
94//!   This also gives CHOICE and ANY, which have no single outer tag to replace,
95//!   a retaggable envelope.
96//!
97//! These constructors use the context-specific class; their `_APPLICATION` and
98//! `_PRIVATE` variants select those classes instead. `Retaggable` lifts through
99//! `Ref`, `Refined`, and `Mapped`, so constraints and value mappings do not prevent
100//! replacing the enclosing format's tag.
101//!
102//! ## A replaceable outer tag: `tag(form)`
103//!
104//! These formats have one replaceable outer class and number. The form selects
105//! whether the start certificate permits primitive, constructed, or both kinds
106//! of identifiers:
107//!
108//! - **`tag(primitive)`**: `Flag ::= BOOLEAN` has universal primitive tag 1 (`01`).
109//!   DER `Payload ::= OCTET STRING` also uses this kind, with identifier `04`.
110//! - **`tag(constructed)`**: `Record ::= SEQUENCE { flag BOOLEAN }` has universal
111//!   constructed tag 16 (`30`). An EXPLICIT envelope also uses this kind:
112//!   `[0] EXPLICIT BOOLEAN` starts with constructed identifier `a0`.
113//! - **`tag(either)`**: BER `Payload ::= OCTET STRING` accepts primitive `04` and
114//!   constructed `24`. After `IMPLICIT(0, PAYLOAD::Fmt)`, the corresponding
115//!   identifiers are `80` and `a0`. The schema handles normalization of the
116//!   serialized form; there is no separate constructed-bit argument.
117//!
118//! The kind describes the schema's accepted forms; it does not change the
119//! schema's parser or enable constructed encodings by itself.
120//!
121//! ## A fixed start domain: `starts(domain)`
122//!
123//! CHOICE and ANY have no single outer tag to replace. Their nominal types are
124//! zero-sized and expose a start-domain certificate, without implementing
125//! `Retaggable`. Both use `starts(domain)`:
126//!
127//! - `Selection ::= CHOICE { flag BOOLEAN, count INTEGER }` can start with `01`
128//!   or `02`, so it uses `starts(asn1_start_mask(false, 0x6u64, 0, 0, 0))`.
129//!   Bits 1 and 2 of the universal-class bitmap are set; the selected alternative
130//!   supplies the identifier, not CHOICE itself.
131//! - `OpenValue ::= ANY` uses `starts(asn1_start_any_non_eoc())`. It can accept,
132//!   for example, an INTEGER TLV beginning with `02` or a SEQUENCE beginning
133//!   with `30`, but not standalone end-of-contents (`00 00`). This is an ordinary
134//!   start-domain expression, not a separate ANY-specific macro kind.
135//!
136//! Start-domain masks describe identifier leading octets conservatively; distinct
137//! high tag numbers can share a bit. See
138//! [`Asn1StartDomain`](crate::asn1::disjoint::Asn1StartDomain) for that limitation.
139//!
140//! # DER and BER
141//!
142//! Both macros implement the codec specifications, `SafeParser`, `SoundParser`,
143//! `Productive`, `NonTailFmt`, `GoodSerializer`, `SPRoundTripDps`, both serializer
144//! equivalence traits, and the executable `Parser`, `Serializer`, `Prepare`, and
145//! `ByteLen` APIs. DER additionally implements `StrictParser`, `NonMalleable`, and
146//! allocation-free `DerOrd`. The inner format must support the delegated traits.
147//!
148//! For a BER schema, use BER combinators in `schema()` and invoke `impl_ber!`:
149//!
150//! ```rust,ignore
151//! vest_lib::impl_ber!(
152//!     tag(constructed), borrowed,
153//!     ITEM, ITEM__, ItemSpec, Item,
154//!     ItemForward, ItemReverse
155//! );
156//! ```
157//!
158//! Changing the macro alone does not change the schema's encoding rules. BER
159//! does not generally preserve encoded length or have a unique encoding, so its
160//! macro does not add `StrictParser`, `NonMalleable`, or `DerOrd` implementations.
161//!
162//! # Proof boundaries and maintenance
163//!
164//! The nominal specifications are opaque. Proofs reveal only the operations they
165//! delegate, and establish the inner invariants locally. The nominal invariants
166//! retain their default `true` definitions: parents use the proved properties
167//! without rechecking the child's combinator tree.
168//!
169//! Mapped BER parsing also hides the closure-backed wire parser, while leaving
170//! the structural `BiMap` visible. This is deliberately narrower than hiding all
171//! `Mapped` parsers or child formats. Scoped equality proofs in `parse` and
172//! `lemma_serialize_len` likewise avoid exposing a whole schema to the solver.
173//!
174//! Implementation responsibilities are separated into four files:
175//!
176//! - `schema.rs`: inner constructors, retagging, and start-domain certificates.
177//! - `proofs.rs`: opaque specifications and common / DER / BER proof traits.
178//! - `exec.rs`: shared borrowed/owned executable delegation.
179//! - `der_ord.rs`: shared borrowed/owned DER ordering delegation.
180//!
181//! This file only assembles those pieces. Internal helper macros are exported
182//! because public macros must resolve them from other crates; they are not a
183//! supported interface. Changes should be checked against the ASN.1 width,
184//! depth, mixed-rule, and CMS fixtures, not just a primitive example.
185mod der_ord;
186mod exec;
187mod proofs;
188mod schema;

190/// Implements the verified DER traits and executable APIs for a generated nominal format.
191///
192/// See [`asn1::macros`](crate::asn1::macros) for the required schema declarations,
193/// invocation arguments, and examples.
194#[macro_export]
195macro_rules! impl_der {
196    (
197        $kind:ident($($argument:tt)*), borrowed,
198        $fmt:ident, $inner:ty, $spec:ty, $value:ident $(, $forward:ty, $reverse:ty)?
199    ) => {
200        $crate::__impl_asn1_nominal_inner!($kind($($argument)*), $fmt, $inner);
201        $crate::__impl_asn1_nominal_specs_and_proofs!($fmt, $spec);
202        $crate::__impl_asn1_nominal_exec!(borrowed, $fmt, $value);
203        $crate::__impl_asn1_nominal_der_ord!(
204            borrowed, $fmt, $inner, $spec, $value $(, $forward, $reverse)?
205        );
206        $crate::__impl_asn1_nominal_tag_support!($kind($($argument)*), $fmt);
207        $crate::impl_der!(
208            @parser_proofs $kind($($argument)*), $fmt, $spec $(, $forward, $reverse)?
209        );
210    };
211    (
212        $kind:ident($($argument:tt)*), owned,
213        $fmt:ident, $inner:ty, $spec:ty, $value:ty $(, $forward:ty, $reverse:ty)?
214    ) => {
215        $crate::__impl_asn1_nominal_inner!($kind($($argument)*), $fmt, $inner);
216        $crate::__impl_asn1_nominal_specs_and_proofs!($fmt, $spec);
217        $crate::__impl_asn1_nominal_exec!(owned, $fmt, $value);
218        $crate::__impl_asn1_nominal_der_ord!(
219            owned, $fmt, $inner, $spec, $value $(, $forward, $reverse)?
220        );
221        $crate::__impl_asn1_nominal_tag_support!($kind($($argument)*), $fmt);
222        $crate::impl_der!(
223            @parser_proofs $kind($($argument)*), $fmt, $spec $(, $forward, $reverse)?
224        );
225    };
226    // Retagged DER formats may have an invalid expected tag. Establish their
227    // inner invariants only after parsing succeeds; other formats need no guard.
228    (@parser_proofs tag($form:ident), $fmt:ident, $spec:ty $(, $forward:ty, $reverse:ty)?) => {
229        $crate::__impl_asn1_nominal_der_tagged_proofs!($fmt, $spec $(, $forward, $reverse)?);
230    };
231    (@parser_proofs starts($domain:expr), $fmt:ident, $spec:ty $(, $forward:ty, $reverse:ty)?) => {
232        $crate::__impl_asn1_nominal_der_fixed_proofs!($fmt, $spec $(, $forward, $reverse)?);
233    };
234}

243        $kind:ident($($argument:tt)*), borrowed,
244        $fmt:ident, $inner:ty, $spec:ty, $value:ident $(, $forward:ty, $reverse:ty)?
245    ) => {

249        $crate::__impl_asn1_nominal_ber_sound!($fmt, $spec $(, $forward, $reverse)?);
250        $crate::__impl_asn1_nominal_tag_support!($kind($($argument)*), $fmt);
251    };
252    (
253        $kind:ident($($argument:tt)*), owned,
254        $fmt:ident, $inner:ty, $spec:ty, $value:ty $(, $forward:ty, $reverse:ty)?
255    ) => {
256        $crate::__impl_asn1_nominal_inner!($kind($($argument)*), $fmt, $inner);
