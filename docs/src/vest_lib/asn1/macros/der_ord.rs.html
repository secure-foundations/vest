1//! Allocation-free DER ordering, delegated to the inner format's cursor.
2//!
3//! Borrowed and owned values differ only in the impl lifetime. Only the reverse
4//! mapper is revealed: ordering observes serialization, never parsing.
5#[doc(hidden)]
6#[macro_export]
7macro_rules! __impl_asn1_nominal_der_ord {
8    (borrowed, $fmt:ident, $inner:ty, $spec:ty, $value:ident $(, $forward:ty, $reverse:ty)?) => {
9        $crate::__impl_asn1_nominal_der_ord!(@impl ['i], $fmt, $inner, $spec, $value<'i> $(, $forward, $reverse)?);
10    };
11    (owned, $fmt:ident, $inner:ty, $spec:ty, $value:ty $(, $forward:ty, $reverse:ty)?) => {
12        $crate::__impl_asn1_nominal_der_ord!(@impl [], $fmt, $inner, $spec, $value $(, $forward, $reverse)?);
13    };
14    // [value lifetime] is empty for owned values; the delegated cursor is unchanged.
15    (@impl [$($value_lt:lifetime)?], $fmt:ident, $inner:ty, $spec:ty, $value:ty $(, $forward:ty, $reverse:ty)?) => {
16        verus! {
17
18        impl DerState for $fmt {
19            type State = <$inner as DerState>::State;
20        }
21
22        impl$(<$value_lt>)? DerOrd<$value> for $fmt {
23            proof fn lemma_der_serialize_len(&self, value: $spec) {
24                $(
25                    reveal(<$reverse as $crate::combinators::mapped::spec::SpecMap>::spec_map);
26                )?
27                reveal(<$fmt as Consistency>::consistent);
28                reveal(<$fmt as SpecSerializer>::spec_serialize);
29                reveal(<$fmt as SpecByteLen>::byte_len);
30                <$inner as DerOrd<$value>>::lemma_der_serialize_len(&self.spec_inner(), value);
31            }
32
33            open spec fn der_remaining(
34                &self,
35                value: $spec,
36                state: <Self as DerState>::State,
37            ) -> Seq<u8> {
38                <$inner as DerOrd<$value>>::der_remaining(&self.spec_inner(), value, state)
39            }
40
41            open spec fn der_state_valid(
42                &self,
43                value: $spec,
44                state: <Self as DerState>::State,
45            ) -> bool {
46                <$inner as DerOrd<$value>>::der_state_valid(&self.spec_inner(), value, state)
47            }
48
49            fn der_start(&self, value: &$value) -> (state: <Self as DerState>::State) {
50                proof {
51                    $(
52                        reveal(<$reverse as $crate::combinators::mapped::spec::SpecMap>::spec_map);
53                    )?
54                    reveal(<$fmt as Consistency>::consistent);
55                    reveal(<$fmt as SpecSerializer>::spec_serialize);
56                }
57                let inner = self.exec_inner();
58                <$inner as DerOrd<$value>>::der_start(&inner, value)
59            }
60
61            fn der_next(
62                &self,
63                value: &$value,
64                state: &mut <Self as DerState>::State,
65            ) -> (next: Option<u8>) {
66                proof {
67                    $(
68                        reveal(<$reverse as $crate::combinators::mapped::spec::SpecMap>::spec_map);
69                    )?
70                    reveal(<$fmt as Consistency>::consistent);
71                }
72                let inner = self.exec_inner();
73                <$inner as DerOrd<$value>>::der_next(&inner, value, state)
74            }
75        }
76
77        } // verus!
78    };
79}
