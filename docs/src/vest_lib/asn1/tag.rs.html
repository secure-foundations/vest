149pub open spec fn tag_number_wf(num: TagNumber) -> bool {
150    num matches TagNumber::Other { tag_num } ==> {
151        &&& !matches!(tag_num, 0 | 1 | 2 | 3 | 4 | 5 | 6 | 9 | 10 | 12 | 13 | 16 | 17 | 18 | 19 | 20 | 21
152            | 22 | 23 | 24 | 26 | 27 | 28 | 30)
153        &&& nat_to_base128(tag_num as nat).len() <= BASE128_MAX_BYTES
154    }

270pub open spec fn class_of_first_byte(b1: u8) -> Class {
271    match b1 & TAG_CLASS_MASK {
272        0b0000_0000u8 => Class::Universal,
273        0b0100_0000u8 => Class::Application,
274        0b1000_0000u8 => Class::ContextSpecific,
275        _ => Class::Private,
276    }

284        Class::Private => 0b1100_0000u8,

288pub open spec fn constructed_of_first_byte(b1: u8) -> bool {

292pub open spec fn constructed_bit(constructed: bool) -> u8 {

300pub open spec fn first_byte_from_parts(class: Class, constructed: bool, low_bits: u8) -> u8 {
301    class_bits(class) | constructed_bit(constructed) | (low_bits & TAG_NUMBER_MASK)

304type TagWireFmt = Bind<U8, spec_fn(u8) -> Sum<Empty, Refined<Base128Fmt<true>, PredFnSpec<UInt>>>>;
305
306type TagInnerFmt = Mapped<TagWireFmt, FnSpecMapper<(u8, Sum<(), UInt>), Tag>>;
307
308#[verusfmt::skip]
309pub(crate) open(crate) spec fn tag_wire() -> TagWireFmt {
310    Bind(U8, |b1: u8| {
311        if b1 & TAG_NUMBER_MASK == TAG_LONG_FORM_SENTINEL {
312            R(Refined(Base128Fmt::<true>, |n: UInt| n >= TAG_LONG_FORM_SENTINEL as UInt))
313        } else {
314            L(Empty)
315        }
316    })
317}
318
319pub(crate) open(crate) spec fn tag_fmt() -> TagInnerFmt {
320    Mapped {
321        inner: tag_wire(),
322        mapper: (
323            |r: (u8, Sum<(), UInt>)|
324                {
325                    let (b1, rest) = r;
326                    let num = match rest {
327                        L(()) => (b1 & TAG_NUMBER_MASK) as UInt,
328                        R(n) => n,
329                    };
330                    Tag {
331                        class: class_of_first_byte(b1),
332                        constructed: constructed_of_first_byte(b1),
333                        number: uint_to_tag_num(num),
334                    }
335                },
336            |tag: Tag|
337                {
338                    let num = tag_num_to_uint(tag.number);
339                    if num < TAG_LONG_FORM_SENTINEL as UInt {
340                        (first_byte_from_parts(tag.class, tag.constructed, num as u8), L(()))
341                    } else {
342                        (
343                            first_byte_from_parts(
344                                tag.class,
345                                tag.constructed,
346                                TAG_LONG_FORM_SENTINEL,
347                            ),
348                            R(num),
349                        )
350                    }
351                },
352        ),
353    }
354}

356// ── Bit-vector helpers ────────────────────────────────────────────────────────
357proof fn lemma_class_bits_roundtrip(b1: u8)
358    ensures
359        class_bits(class_of_first_byte(b1)) == (b1 & TAG_CLASS_MASK),
360{
361    let cls = b1 & TAG_CLASS_MASK;
362    assert({
363        ||| cls == 0b0000_0000u8
364        ||| cls == 0b0100_0000u8
365        ||| cls == 0b1000_0000u8
366        ||| cls == 0b1100_0000u8
367    }) by (bit_vector)
368        requires
369            cls == (b1 & TAG_CLASS_MASK),
370    ;
371}
372
373proof fn lemma_class_bits_only_class_mask(class: Class)
374    ensures
375        class_bits(class) & TAG_CONSTRUCTED_MASK == 0u8,
376        class_bits(class) & TAG_NUMBER_MASK == 0u8,
377        class_bits(class) & TAG_CLASS_MASK == class_bits(class),
378{
379    assert(forall|cls: u8|
380        {
381            ||| cls == 0b0000_0000u8
382            ||| cls == 0b0100_0000u8
383            ||| cls == 0b1000_0000u8
384            ||| cls == 0b1100_0000u8
385        } ==> (cls & TAG_CONSTRUCTED_MASK == 0u8 && cls & TAG_NUMBER_MASK == 0u8 && (cls
386            & TAG_CLASS_MASK) == cls)) by (bit_vector);
387}
388
389proof fn lemma_first_byte_from_parts_roundtrip(class: Class, constructed: bool, low_bits: u8)
390    ensures
391        class_of_first_byte(first_byte_from_parts(class, constructed, low_bits)) == class,
392        constructed_of_first_byte(first_byte_from_parts(class, constructed, low_bits))
393            == constructed,
394        first_byte_from_parts(class, constructed, low_bits) & TAG_NUMBER_MASK == low_bits
395            & TAG_NUMBER_MASK,
396{
397    let fb = first_byte_from_parts(class, constructed, low_bits);
398    let cb = class_bits(class);
399    let cbit = constructed_bit(constructed);
400    lemma_class_bits_only_class_mask(class);
401
402    assert(fb & TAG_CLASS_MASK == cb && (fb & TAG_CONSTRUCTED_MASK != 0u8) == constructed && fb
403        & TAG_NUMBER_MASK == low_bits & TAG_NUMBER_MASK) by (bit_vector)
404        requires
405            fb == cb | cbit | (low_bits & TAG_NUMBER_MASK),
406            cb & TAG_CONSTRUCTED_MASK == 0u8,
407            cb & TAG_NUMBER_MASK == 0u8,
408            (cb & TAG_CLASS_MASK) == cb,
409            cbit == TAG_CONSTRUCTED_MASK || cbit == 0u8,
410            constructed ==> cbit == TAG_CONSTRUCTED_MASK,
411            !constructed ==> cbit == 0u8,
412    ;
413}

415proof fn lemma_first_byte_roundtrip(b1: u8)
416    ensures
417        first_byte_from_parts(
418            class_of_first_byte(b1),
419            constructed_of_first_byte(b1),
420            b1 & TAG_NUMBER_MASK,
421        ) == b1,
422{
423    lemma_class_bits_roundtrip(b1);
424    let fb = first_byte_from_parts(
425        class_of_first_byte(b1),
426        constructed_of_first_byte(b1),
427        b1 & TAG_NUMBER_MASK,
428    );
429    let cb = class_bits(class_of_first_byte(b1));
430    assert(fb == b1) by (bit_vector)
431        requires
432            fb == cb | constructed_bit(constructed_of_first_byte(b1)) | ((b1 & TAG_NUMBER_MASK)
433                & TAG_NUMBER_MASK),
434            cb == (b1 & TAG_CLASS_MASK),
435    ;
436}

438proof fn lemma_tag_fmt_sound_nonmal_inv()
439    ensures
440        tag_fmt().sound_inv(),
441        tag_fmt().nonmal_inv(),
442{
443    let fmt = tag_fmt();
444    assert forall|v| fmt.inner.consistent(v) implies (fmt.mapper.1)((fmt.mapper.0)(v)) == v by {
445        let (b1, rest) = v;
446        lemma_first_byte_roundtrip(b1);
447        if b1 & TAG_NUMBER_MASK == TAG_LONG_FORM_SENTINEL {
448        } else {
449            let num = (b1 & TAG_NUMBER_MASK) as UInt;
450            assert(num < TAG_LONG_FORM_SENTINEL as UInt) by (bit_vector)
451                requires
452                    (b1 & TAG_NUMBER_MASK) != TAG_LONG_FORM_SENTINEL,
453                    num == (b1 & TAG_NUMBER_MASK) as UInt,
454            ;
455        }
456    }

459proof fn lemma_tag_fmt_unambiguous(tag: Tag)
460    requires
461        tag_fmt().consistent(tag),
462        tag_number_wf(tag.number),
463    ensures
464        (tag_fmt().mapper.0)((tag_fmt().mapper.1)(tag)) == tag,
465{
466    let num = tag_num_to_uint(tag.number);
467    if num < TAG_LONG_FORM_SENTINEL as UInt {
468        let low = num as u8;
469        lemma_first_byte_from_parts_roundtrip(tag.class, tag.constructed, low);
470        assert(low & TAG_NUMBER_MASK == low) by (bit_vector)
471            requires
472                low == num as u8,
473                num < TAG_LONG_FORM_SENTINEL as UInt,
474        ;
475    } else {
476        lemma_first_byte_from_parts_roundtrip(tag.class, tag.constructed, TAG_LONG_FORM_SENTINEL);
477    }
478}

480proof fn lemma_tag_wf_implies_tag_fmt_consistent(tag: Tag)
481    requires
482        tag_number_wf(tag.number),
483    ensures
484        tag_fmt().consistent(tag),
485{
486    let num = tag_num_to_uint(tag.number);
487    assert(TAG_LONG_FORM_SENTINEL & TAG_NUMBER_MASK == TAG_LONG_FORM_SENTINEL) by (bit_vector);
488    if num < TAG_LONG_FORM_SENTINEL as UInt {
489        let low = num as u8;
490        lemma_first_byte_from_parts_roundtrip(tag.class, tag.constructed, low);
491        assert(low & TAG_NUMBER_MASK == low) by (bit_vector)
492            requires
493                low == num as u8,
494                num < TAG_LONG_FORM_SENTINEL as UInt,
495        ;
496    } else {
497        lemma_first_byte_from_parts_roundtrip(tag.class, tag.constructed, TAG_LONG_FORM_SENTINEL);
498        lemma_base128_fmt_consistent::<true>(num);
499    }

646impl Parser<&[u8]> for super::TagFmt {
647    type PT = Tag;
648
649    fn parse(&self, ibuf: &&[u8]) -> PResult<Self::PT> {
650        broadcast use crate::core::spec::SafeParser::lemma_parse_safe;
651        broadcast use crate::core::spec::SoundParser::lemma_parse_sound_value;
652
653        let _ = ibuf.len();
654
655        let (n1, b1): (usize, u8) = U8.parse(ibuf)?;
656        let rest = ibuf.skip(n1);
657
658        let (n2, num) = if b1 & TAG_NUMBER_MASK == TAG_LONG_FORM_SENTINEL {
659            let (n2, num) = Base128Fmt::<true>.parse(&rest)?;
660            if num < TAG_LONG_FORM_SENTINEL as UInt {
661                return Err(ParseError::non_canonical());
662            }
663            (n2, num)
664        } else {
665            (0, (b1 & TAG_NUMBER_MASK) as UInt)
666        };
667
668        let class = match b1 & TAG_CLASS_MASK {
669            0b0000_0000u8 => Class::Universal,
670            0b0100_0000u8 => Class::Application,
671            0b1000_0000u8 => Class::ContextSpecific,
672            _ => Class::Private,
673        };
674        let constructed = b1 & TAG_CONSTRUCTED_MASK != 0;
675        let number = match num {
676            0 => TagNumber::EOC,
677            1 => TagNumber::Boolean,
678            2 => TagNumber::Integer,
679            3 => TagNumber::BitString,
680            4 => TagNumber::OctetString,
681            5 => TagNumber::Null,
682            6 => TagNumber::ObjectIdentifier,
683            9 => TagNumber::Real,
684            10 => TagNumber::Enumerated,
685            12 => TagNumber::Utf8String,
686            13 => TagNumber::RelativeOid,
687            16 => TagNumber::Sequence,
688            17 => TagNumber::Set,
689            18 => TagNumber::NumericString,
690            19 => TagNumber::PrintableString,
691            20 => TagNumber::TeletexString,
692            21 => TagNumber::VideotexString,
693            22 => TagNumber::Ia5String,
694            23 => TagNumber::UtcTime,
695            24 => TagNumber::GeneralizedTime,
696            26 => TagNumber::VisibleString,
697            27 => TagNumber::GeneralString,
698            28 => TagNumber::UniversalString,
699            30 => TagNumber::BmpString,
700            other => TagNumber::Other { tag_num: other },
701        };
702
703        Ok((n1 + n2, Tag { class, constructed, number }))
704    }
705}

707impl<Output: OutputBuf> Serializer<Output, Tag> for super::TagFmt {
708    fn serialize_into(&self, v: &Tag, obuf: &mut Output) {
709        broadcast use crate::core::exec::output::outbuf_lemmas;
710
711        let num = tag_number_value(v.number);
712
713        let class_bits = match v.class {
714            Class::Universal => 0b0000_0000u8,
715            Class::Application => 0b0100_0000u8,
716            Class::ContextSpecific => 0b1000_0000u8,
717            Class::Private => 0b1100_0000u8,
718        };
719        let constructed_bit = if v.constructed {
720            TAG_CONSTRUCTED_MASK
721        } else {
722            0u8
723        };
724        proof {
725            lemma_tag_wf_implies_tag_fmt_consistent(*v);
726        }
727        if num < TAG_LONG_FORM_SENTINEL as UInt {
728            let low = num as u8;
729            let b1 = class_bits | constructed_bit | (low & TAG_NUMBER_MASK);
730            U8.serialize_into(&b1, obuf);
731        } else {
732            let b1 = class_bits | constructed_bit | TAG_LONG_FORM_SENTINEL & TAG_NUMBER_MASK;
733            U8.serialize_into(&b1, obuf);
734            Base128Fmt::<true>.serialize_into(&num, obuf);
735        }
736    }
737}

739impl Prepare<Tag> for super::TagFmt {
740    fn prepare(&self, v: &Tag) -> Result<usize, PreSerializeError> {
741        if let TagNumber::Other { tag_num } = v.number {
742            if matches!(tag_num, 0 | 1 | 2 | 3 | 4 | 5 | 6 | 9 | 10 | 12 | 13 | 16 | 17 | 18 | 19 | 20 | 21
743                | 22 | 23 | 24 | 26 | 27 | 28 | 30) {
744                return Err(PreSerializeError::custom("Invalid tag number"));
745            }
746        }
747        let num = tag_number_value(v.number);
748
749        proof {
750            lemma_to_base128_len_bounds();
751            lemma_base128_fmt_byte_len::<true>(num);
752        }
753        let nbytes = Base128Fmt::<true>.length(&num);
754        if nbytes > BASE128_MAX_BYTES {
755            return Err(PreSerializeError::length_too_large());
756        }
757        proof {
758            assert(tag_number_wf(v.deep_view().number));
759            lemma_tag_wf_implies_tag_fmt_consistent(v.deep_view());
760        }
761
762        if num < TAG_LONG_FORM_SENTINEL as UInt {
763            Ok(1)
764        } else {
765            Ok(1 + nbytes)
766        }
767    }
768}
769
770impl ByteLen<Tag> for super::TagFmt {
771    fn length(&self, v: &Tag) -> usize {
772        let num = tag_number_value(v.number);
773
774        proof {
775            lemma_to_base128_len_bounds();
776            lemma_base128_fmt_byte_len::<true>(num);
777            lemma_first_byte_from_parts_roundtrip(v.class, v.constructed, TAG_LONG_FORM_SENTINEL);
778            assert(TAG_LONG_FORM_SENTINEL & TAG_NUMBER_MASK == TAG_LONG_FORM_SENTINEL)
779                by (bit_vector);
780        }
781        let nbytes = Base128Fmt::<true>.length(&num);
782
783        if num < TAG_LONG_FORM_SENTINEL as UInt {
784            1
785        } else {
786            1 + nbytes
787        }
788    }
789}
