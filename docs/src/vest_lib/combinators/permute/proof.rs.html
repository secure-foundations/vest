42impl<P1: SoundParser, P2: SoundParser> SoundParser for super::Permute2<P1, P2> {
43    open spec fn sound_inv(&self) -> bool {
44        &&& self.0.sound_inv()
45        &&& self.1.sound_inv()
46    }
47
48    proof fn lemma_parse_sound_consumption(&self, ibuf: Seq<u8>) {
49        let canonical = Pair(self.0, self.1);
50        let swapped = Pair(self.1, self.0);
51        canonical.lemma_parse_sound_consumption(ibuf);
52        swapped.lemma_parse_sound_consumption(ibuf);
53        // Either the declared order matched, in which case the length claim is `Pair`'s, or the
54        // swapped order matched and the two length sums agree by commutativity of `nat` addition.
55        if canonical.spec_parse(ibuf) is None {
56            if let Some((_n, iv)) = swapped.spec_parse(ibuf) {
57                assert(self.byte_len(super::swap2(iv)) == swapped.byte_len(iv));
58            }
59        }
60    }
61
62    proof fn lemma_parse_sound_value(&self, ibuf: Seq<u8>) {
63        let canonical = Pair(self.0, self.1);
64        let swapped = Pair(self.1, self.0);
65        canonical.lemma_parse_sound_value(ibuf);
66        swapped.lemma_parse_sound_value(ibuf);
67        // Both branches establish the same conjunction of component consistencies, just reordered.
68        if canonical.spec_parse(ibuf) is None {
69            if let Some((_n, iv)) = swapped.spec_parse(ibuf) {
70                assert(self.consistent(super::swap2(iv)));
71            }
72        }
73    }
74}
