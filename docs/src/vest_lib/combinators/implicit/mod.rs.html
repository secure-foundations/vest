12/// A family of dependent combinators indexed by a key type.
13///
14/// Used to constrain the `Tail` combinator in a [`Implicit<Head, Tail>`].
15pub trait DepCombinator {
16    /// The type of keys parsed by the head combinator/to be recovered during serialization.
17    type Key;
18
19    /// The type of values consumed/produced by the body combinator.
20    type Val;
21
22    /// The type of the body combinator produced by `apply`.
23    type Body: Consistency<Val = Self::Val>;
24
25    /// Given a key, produce the body combinator for that key.
26    spec fn apply(&self, key: Self::Key) -> Self::Body;
27
28    /// Given a body value, recover the key used to produce the body combinator.
29    spec fn recover(&self, value: Self::Val) -> Self::Key;
30
31    open spec fn recover_inv(&self) -> bool {
32        true
33    }
34
35    /// Recover must agree with any key that makes the body's value consistent.
36    proof fn lemma_recover_consistent(&self, key: Self::Key, value: Self::Val)
37        requires
38            self.recover_inv(),
39        ensures
40            self.apply(key).consistent(value) ==> self.recover(value) == key,
41    ;
42}
43
44/// Dependent family encoded by a pair of pure spec closures `(apply, recover)`.
45pub type KVFormat<Key, Value, Fmt> = (spec_fn(Key) -> Fmt, spec_fn(Value) -> Key);

47/// Dependent sequential combinator with deterministic key recovery.
48///
49/// Parsing semantics: parse `Head` to get a key, then parse the
50/// body via `Tail::apply(key)`. Only the body value is returned.
51///
52/// The key is recovered via `Tail::recover(value)` during serialization.
53///
54/// ## Consistency
55///
56/// A value `v: Tail::Val` is consistent with `Implicit(Head, Tail)` iff
57/// ```text
58/// let key = self.1.recover(v);
59/// self.0.consistent(key) && self.1.apply(key).consistent(v)
60/// ```
61///
62/// ## Unambiguity
63///
64/// ```text
65/// self.0.unambiguous() &&
66/// forall|key: Head::PVal| #[trigger] (self.1.apply(key)).unambiguous()
67/// ```
68#[derive(Copy)]
69pub struct Implicit<Head, Tail>(pub Head, pub Tail);
70
71impl<Head: Clone, Tail: Clone> Clone for Implicit<Head, Tail> {
72    fn clone(&self) -> (cloned: Self)
73        ensures
74            call_ensures(Head::clone, (&self.0,), cloned.0),
75            call_ensures(Tail::clone, (&self.1,), cloned.1),
76    {
77        Implicit(self.0.clone(), self.1.clone())
78    }
79}
