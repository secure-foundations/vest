191impl<A, Then> NonMalleable for super::AndThen<A, Then> where
192    A: BytesCombinator + SoundParser + NonMalleable,
193    Then: NonMalleable,
194 {
195    open spec fn nonmal_inv(&self) -> bool {
196        &&& self.0.nonmal_inv()
197        &&& self.0.sound_inv()
198        &&& self.1.nonmal_inv()
199        &&& self.1.safe_inv()
200    }
201
202    proof fn lemma_parse_non_malleable(&self, buf1: Seq<u8>, buf2: Seq<u8>) {
203        if let Some((n1, v1)) = self.spec_parse(buf1) {
204            if let Some((n2, v2)) = self.spec_parse(buf2) {
205                if v1 == v2 {
206                    let (n1a, chunk1) = self.0.spec_parse(buf1)->0;
207                    let (n2a, chunk2) = self.0.spec_parse(buf2)->0;
208                    let (n1b, v1b) = self.1.spec_parse(chunk1)->0;
209                    let (n2b, v2b) = self.1.spec_parse(chunk2)->0;
210                    self.0.lemma_parse_sound_consumption(buf1);
211                    self.0.lemma_parse_sound_consumption(buf2);
212                    self.0.lemma_byte_len_is_buf_len(chunk1);
213                    self.0.lemma_byte_len_is_buf_len(chunk2);
214                    self.0.lemma_parse_non_malleable(buf1, buf2);
215                    self.1.lemma_parse_non_malleable(chunk1, chunk2);
216                    assert(n1 == n1a && n2 == n2a);
217                    assert(chunk1.take(n1a) == chunk2.take(n2a));
218                    assert(chunk1.take(n1a) == chunk1);
219                    assert(chunk2.take(n2a) == chunk2);
220                    assert(chunk1 == chunk2);
221                    assert(buf1.take(n1a) == buf2.take(n2a));
222                }
223            }
224        }
225    }
226}

228impl<A, Then> NoLookAhead for super::AndThen<A, Then> where
229    A: BytesCombinator + NoLookAhead<PVal = Seq<u8>>,
230    Then: SafeParser,
231 {
232    open spec fn no_lookahead_inv(&self) -> bool {
233        self.0.no_lookahead_inv()
234    }
235
236    proof fn lemma_no_lookahead(&self, i1: Seq<u8>, i2: Seq<u8>) {
237        if let Some((n, v)) = self.spec_parse(i1) {
238            if 0 <= n <= i2.len() {
239                if i2.take(n) == i1.take(n) {
240                    self.0.lemma_no_lookahead(i1, i2);
241                }
242            }
243        }
244    }
245}
