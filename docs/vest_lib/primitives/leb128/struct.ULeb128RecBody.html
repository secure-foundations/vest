open spec fn spec_body( &self, _param: (), rec: ParamRecSpecs<Self::Param, Self::T>, ) -> Self::Body

{
    Alt(
        terminal_byte_nat(),
        Mapped {
            inner: Pair(
                continuation_byte(),
                Refined(rec(()), |v: nat| MINIMAL ==> v > 0),
            ),
            mapper: (
                |pair: (u8, nat)| 128 * pair.1 + pair.0 as nat,
                |o: nat| ((o % 128) as u8, o / 128),
            ),
        },
    )
}
