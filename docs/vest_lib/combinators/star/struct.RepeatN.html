pub open spec fn parse_n_rec( &self, count: nat, ibuf: Seq<u8>, ) -> Option<(int, Seq<C::PVal>)>

{
    if count == 0 {
        Some((0, Seq::empty()))
    } else {
        match self.1.spec_parse(ibuf) {
            Some((n0, v0)) => {
                match self.parse_n_rec((count - 1) as nat, ibuf.skip(n0)) {
                    Some((n1, vs1)) => Some((n0 + n1, seq![v0] + vs1)),
                    None => None,
                }
            }
            None => None,
        }
    }
}
